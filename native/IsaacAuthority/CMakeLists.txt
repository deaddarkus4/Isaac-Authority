    add_library(IsaacAuthorityInputPredict SHARED src/input_adapter.cpp input_predict.def)

    target_link_libraries(IsaacAuthorityInputPredict PRIVATE authority_game_support input_core ws2_32)
    # Stepped: one command per step of the driven player. The host plays commands out of a small buffer instead of reading
    # the newest one, and the predicting client captures one command per step of its own player.
    add_library(IsaacAuthorityInputSteppedHost SHARED src/input_adapter.cpp input_stepped_host.def)
    target_compile_definitions(IsaacAuthorityInputSteppedHost PRIVATE ISAAC_INPUT_STEPPED)

    target_compile_definitions(IsaacAuthorityInputSteppedClient PRIVATE ISAAC_INPUT_PREDICT ISAAC_INPUT_STEPPED)
    target_link_libraries(IsaacAuthorityInputSteppedClient PRIVATE authority_game_support input_core ws2_32)
    add_library(IsaacAuthorityNative SHARED src/native_adapter.cpp native_adapter.def)
    target_link_libraries(IsaacAuthorityNative PRIVATE authority_game_support ws2_32 shell32)

    if(CMAKE_SIZEOF_VOID_P EQUAL 4)
        add_library(IsaacAuthorityVersionProxy SHARED src/version_proxy.cpp version_proxy.def)
        set_target_properties(IsaacAuthorityVersionProxy PROPERTIES OUTPUT_NAME version RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/proxy" ARCHIVE_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/proxy")
        target_compile_options(IsaacAuthorityVersionProxy PRIVATE /W4 /WX /permissive- /utf-8)
    endif()
    add_library(IsaacAuthorityLocalhost SHARED src/localhost_adapter.cpp localhost_adapter.def)
    target_link_libraries(IsaacAuthorityLocalhost PRIVATE authority_game_support)

    add_executable(world_tests tests/world_tests.cpp)
    target_link_libraries(world_tests PRIVATE world_receiver)
endif()

    endif()
