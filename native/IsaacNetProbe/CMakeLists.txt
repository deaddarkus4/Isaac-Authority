cmake_minimum_required(VERSION 3.20)
project(IsaacNetProbe VERSION 0.2.0 LANGUAGES CXX)
if(NOT WIN32 OR NOT CMAKE_SIZEOF_VOID_P EQUAL 4)
    message(FATAL_ERROR "IsaacNetProbe requires Windows x86. Configure with -A Win32.")
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
add_library(probe_core STATIC src/profile.cpp)
target_include_directories(probe_core PUBLIC include)
target_compile_definitions(probe_core PUBLIC WIN32_LEAN_AND_MEAN NOMINMAX)
target_link_libraries(probe_core PUBLIC bcrypt)
add_library(IsaacNetProbe SHARED src/dll.cpp exports.def)
add_library(probe_recorder STATIC src/recorder.cpp)
target_link_libraries(probe_recorder PUBLIC probe_core)
target_link_libraries(IsaacNetProbe PRIVATE probe_recorder)
add_executable(IsaacNetProbeCli src/cli.cpp)
target_link_libraries(IsaacNetProbeCli PRIVATE probe_core)
add_executable(probe_tests tests/probe_tests.cpp)
target_link_libraries(probe_tests PRIVATE probe_core)
add_executable(recorder_tests tests/recorder_tests.cpp)
target_link_libraries(recorder_tests PRIVATE probe_recorder)
add_library(probe_injector STATIC src/injector.cpp)
target_link_libraries(probe_injector PUBLIC probe_core)
add_executable(IsaacNetProbeAttach src/attach.cpp)
target_link_libraries(IsaacNetProbeAttach PRIVATE probe_injector)
add_executable(probe_fixture_host tests/fixture_host.cpp)
add_library(ProbeRecorderFixture SHARED tests/fixture_dll.cpp tests/fixture_exports.def)
target_link_libraries(ProbeRecorderFixture PRIVATE probe_recorder)
add_executable(injection_tests tests/injection_tests.cpp)
target_link_libraries(injection_tests PRIVATE probe_injector)
foreach(target probe_core probe_recorder probe_injector IsaacNetProbe IsaacNetProbeCli IsaacNetProbeAttach probe_tests recorder_tests probe_fixture_host ProbeRecorderFixture injection_tests)
    if(MSVC)
        target_compile_options(${target} PRIVATE /W4 /WX /permissive- /utf-8)
    endif()
endforeach()
enable_testing()
add_test(NAME reject_invalid_images COMMAND probe_tests)
add_test(NAME dll_contract COMMAND probe_tests --dll $<TARGET_FILE:IsaacNetProbe>)
add_test(NAME event_recorder COMMAND recorder_tests)
add_test(NAME remote_recording COMMAND injection_tests $<TARGET_FILE:probe_fixture_host> $<TARGET_FILE:ProbeRecorderFixture>)
set(ISAAC_TEST_EXECUTABLE "" CACHE FILEPATH "Optional original J460 executable for read-only validation")
if(ISAAC_TEST_EXECUTABLE)
    add_test(NAME original_j460 COMMAND probe_tests --game "${ISAAC_TEST_EXECUTABLE}")
endif()
