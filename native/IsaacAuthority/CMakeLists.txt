cmake_minimum_required(VERSION 3.20)
project(IsaacAuthority VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
add_library(authority_core STATIC src/authority.cpp src/wire.cpp)
add_library(snapshot_bridge STATIC src/snapshot_bridge.cpp)
target_include_directories(snapshot_bridge PUBLIC include)
add_library(world_core STATIC src/world_state.cpp)
target_include_directories(world_core PUBLIC include)
target_include_directories(authority_core PUBLIC include)
add_executable(authority_tests tests/authority_tests.cpp)
target_link_libraries(authority_tests PRIVATE authority_core)
if(WIN32)
    if(NOT CMAKE_SIZEOF_VOID_P EQUAL 4)
        message(FATAL_ERROR "J460 game adapter requires x86; configure with -A Win32")
    endif()
    add_executable(IsaacAuthorityLab src/lab.cpp)
    target_compile_definitions(IsaacAuthorityLab PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX)
    target_link_libraries(IsaacAuthorityLab PRIVATE authority_core ws2_32)
    add_library(authority_game_support STATIC ../IsaacNetProbe/src/profile.cpp ../IsaacNetProbe/src/injector.cpp)
    target_include_directories(authority_game_support PUBLIC ../IsaacNetProbe/include include)
    target_compile_definitions(authority_game_support PUBLIC WIN32_LEAN_AND_MEAN NOMINMAX)
    target_link_libraries(authority_game_support PUBLIC bcrypt)
    add_library(IsaacAuthorityGame SHARED src/game_adapter.cpp game_adapter.def)
    target_link_libraries(IsaacAuthorityGame PRIVATE authority_game_support)
    add_library(IsaacAuthoritySource SHARED src/game_adapter.cpp source_adapter.def)
    target_compile_definitions(IsaacAuthoritySource PRIVATE ISAAC_AUTHORITY_SOURCE)
    target_link_libraries(IsaacAuthoritySource PRIVATE authority_game_support)
    add_library(snapshot_receiver STATIC src/snapshot_receiver.cpp)
    target_include_directories(snapshot_receiver PUBLIC include)
    target_compile_definitions(snapshot_receiver PUBLIC WIN32_LEAN_AND_MEAN NOMINMAX)
    target_link_libraries(snapshot_receiver PUBLIC snapshot_bridge ws2_32 bcrypt)
    add_library(IsaacAuthorityNetwork SHARED src/game_adapter.cpp network_adapter.def)
    target_compile_definitions(IsaacAuthorityNetwork PRIVATE ISAAC_AUTHORITY_UDP)
    target_link_libraries(IsaacAuthorityNetwork PRIVATE authority_game_support snapshot_receiver)
    add_library(IsaacAuthorityReplica SHARED src/game_adapter.cpp replica_adapter.def)
    target_compile_definitions(IsaacAuthorityReplica PRIVATE ISAAC_AUTHORITY_UDP)
    target_link_libraries(IsaacAuthorityReplica PRIVATE authority_game_support snapshot_receiver)
    add_executable(IsaacAuthorityAttach src/game_attach.cpp)
    target_link_libraries(IsaacAuthorityAttach PRIVATE authority_game_support)
    add_executable(IsaacAuthorityLaunch src/isolated_launch.cpp)
    target_link_libraries(IsaacAuthorityLaunch PRIVATE authority_game_support)
    add_executable(game_adapter_tests tests/game_adapter_tests.cpp)
    target_link_libraries(game_adapter_tests PRIVATE authority_game_support)
    add_executable(snapshot_tests tests/snapshot_tests.cpp)
    target_link_libraries(snapshot_tests PRIVATE snapshot_receiver)
    add_library(world_receiver STATIC src/world_receiver.cpp)
    target_include_directories(world_receiver PUBLIC include)
    target_compile_definitions(world_receiver PUBLIC WIN32_LEAN_AND_MEAN NOMINMAX)
    target_link_libraries(world_receiver PUBLIC world_core ws2_32 bcrypt)
    add_library(IsaacAuthorityWorldSource SHARED src/world_adapter.cpp world_source.def)
    target_compile_definitions(IsaacAuthorityWorldSource PRIVATE ISAAC_WORLD_SOURCE)
    target_link_libraries(IsaacAuthorityWorldSource PRIVATE authority_game_support world_receiver)
    add_library(IsaacAuthorityWorldReplica SHARED src/world_adapter.cpp world_replica.def)
    target_link_libraries(IsaacAuthorityWorldReplica PRIVATE authority_game_support world_receiver)
    # Same adapter with real room transitions; the verified tear-only modules above keep their behaviour.
    add_library(IsaacAuthorityRoomSource SHARED src/world_adapter.cpp room_source.def)
    target_compile_definitions(IsaacAuthorityRoomSource PRIVATE ISAAC_WORLD_SOURCE ISAAC_WORLD_ROOMS)
    target_link_libraries(IsaacAuthorityRoomSource PRIVATE authority_game_support world_receiver)
    add_library(IsaacAuthorityRoomReplica SHARED src/world_adapter.cpp room_replica.def)
    target_compile_definitions(IsaacAuthorityRoomReplica PRIVATE ISAAC_WORLD_ROOMS)
    target_link_libraries(IsaacAuthorityRoomReplica PRIVATE authority_game_support world_receiver)
    # Rooms plus enemies: the replica corrects its own room-spawned enemies towards the host's state.
    add_library(IsaacAuthorityNpcSource SHARED src/world_adapter.cpp npc_source.def)
    target_compile_definitions(IsaacAuthorityNpcSource PRIVATE ISAAC_WORLD_SOURCE ISAAC_WORLD_ROOMS ISAAC_WORLD_NPCS)
    target_link_libraries(IsaacAuthorityNpcSource PRIVATE authority_game_support world_receiver)
    add_library(IsaacAuthorityNpcReplica SHARED src/world_adapter.cpp npc_replica.def)
    target_compile_definitions(IsaacAuthorityNpcReplica PRIVATE ISAAC_WORLD_ROOMS ISAAC_WORLD_NPCS)
    target_link_libraries(IsaacAuthorityNpcReplica PRIVATE authority_game_support world_receiver)
    # Client input on the host: received commands replace what the game reads from one controller.
    add_library(input_core STATIC src/input_state.cpp)
    target_include_directories(input_core PUBLIC include)
    add_library(IsaacAuthorityInput SHARED src/input_adapter.cpp input_adapter.def)
    target_link_libraries(IsaacAuthorityInput PRIVATE authority_game_support input_core ws2_32)
    add_executable(input_tests tests/input_tests.cpp)
    target_link_libraries(input_tests PRIVATE input_core)
    add_executable(world_tests tests/world_tests.cpp)
    target_link_libraries(world_tests PRIVATE world_receiver)
endif()
foreach(target authority_core snapshot_bridge snapshot_receiver snapshot_tests world_core world_receiver world_tests IsaacAuthorityWorldSource IsaacAuthorityWorldReplica IsaacAuthorityRoomSource IsaacAuthorityRoomReplica IsaacAuthorityNpcSource IsaacAuthorityNpcReplica input_core IsaacAuthorityInput input_tests authority_tests IsaacAuthorityLab authority_game_support IsaacAuthorityGame IsaacAuthoritySource IsaacAuthorityNetwork IsaacAuthorityReplica IsaacAuthorityAttach IsaacAuthorityLaunch game_adapter_tests)
    if(TARGET ${target} AND MSVC)
        target_compile_options(${target} PRIVATE /W4 /WX /permissive- /utf-8)
    endif()
endforeach()
